cmake_minimum_required(VERSION 3.18)
project(bench LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
find_package(Python 3.12 COMPONENTS Interpreter Development.Module Development.SABIModule REQUIRED)
find_package(Dawn CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT)
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(_bench_nanobind STABLE_ABI NB_STATIC _bench_nanobind.cpp)
target_link_libraries(_bench_nanobind PRIVATE dawn::webgpu_dawn)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_bench_pybind11 _bench_pybind11.cpp)
target_link_libraries(_bench_pybind11 PRIVATE dawn::webgpu_dawn)
